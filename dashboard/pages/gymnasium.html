import gymnasium as gym
from src.simulation.env import CogEWEnv

# 1. Instantiate the standard Gymnasium environment
env = CogEWEnv()
print("Observation Space:", env.observation_space)  # Box(28,)
print("Action Space:", env.action_space)            # Box(4,)

# 2. Reset with reproducible seed
obs, info = env.reset(seed=42)

# 3. Step with continuous control action: [AoA_delta, bank_cmd, ew_mode, standoff_release]
action = [0.0, 0.0, 0.8, -1.0]  # Cruise at level attitude with RGPO active
obs, reward, terminated, truncated, info = env.step(action)

print(f"Step Reward: {reward:.2f}, Altitude: {info['altitude_m']:.1f}m, Mach: {info['mach']:.2f}")
